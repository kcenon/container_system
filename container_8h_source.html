1018 // Thread-safe lock guard classes
1019 // Always acquire lock to eliminate TOCTOU vulnerability (see #190)
1020 // Modern mutex implementations have ~20ns overhead for uncontended locks
1021 // which is negligible compared to actual data access costs

1092 // Template implementation for get<T>()
1093 template<typename T>
1094 [[nodiscard]] kcenon::common::Result<T> value_container::get(std::string_view key) const noexcept {
1095 read_lock_guard lock(this);
1096
1097 for (const auto& val : optimized_units_) {
1098 if (val.name == key) {
1099 // Try to extract the value of type T from the variant
1100 if (auto* ptr = std::get_if<T>(&val.data)) {
1101 return kcenon::common::ok(*ptr);
1102 }
1103 // Type mismatch
1104 return kcenon::common::Result<T>(
1105 kcenon::common::error_info{
1106 error_codes::type_mismatch,
1107 error_codes::make_message(error_codes::type_mismatch, key),
1108 "container_system"});
1109 }
1110 }
1111 // Key not found
1112 return kcenon::common::Result<T>(
1113 kcenon::common::error_info{
1114 error_codes::key_not_found,
1115 error_codes::make_message(error_codes::key_not_found, key),
1116 "container_system"});
1117 }

1120 template<typename T>
1121 [[nodiscard]] kcenon::common::VoidResult value_container::set_result(
1122 std::string_view key, T&& data_val) noexcept {
1123 try {
1124 if (key.empty()) {
1125 return kcenon::common::VoidResult(
1126 kcenon::common::error_info{
1127 error_codes::empty_key,
1128 error_codes::make_message(error_codes::empty_key),
1129 "container_system"});
1130 }
1131
1132 optimized_value val;
1133 val.name = std::string(key);
1134 val.data = std::forward<T>(data_val);
1135 val.type = static_cast<value_types>(val.data.index());
1136 set_unit_impl(val);
1137 return kcenon::common::ok();
1138 } catch (const std::bad_alloc&) {
1139 return kcenon::common::VoidResult(
1140 kcenon::common::error_info{
1141 error_codes::memory_allocation_failed,
1142 error_codes::make_message(error_codes::memory_allocation_failed, key),
1143 "container_system"});
1144 } catch (const std::exception& e) {
1145 return kcenon::common::VoidResult(
1146 kcenon::common::error_info{
1147 error_codes::invalid_value,
1148 std::string("Failed to set value: ") + e.what(),
