102 constexpr std::string_view get_message(int code) noexcept
103 {
104 switch (code)
105 {
106 case missing_required: return "Required field is missing";
107 case type_mismatch: return "Field type mismatch";
108 case out_of_range: return "Value is out of range";
109 case invalid_length: return "Invalid length";
110 case pattern_mismatch: return "Pattern mismatch";
111 case not_in_allowed_values: return "Value not in allowed list";
112 case custom_validation_failed: return "Custom validation failed";
113 case nested_validation_failed: return "Nested schema validation failed";
114 default: return error_codes::get_message(code);
115 }
116 }

122 struct validation_error

124 std::string field;
125 std::string message;
126 int code;

128 validation_error() : code(0) {}

130 validation_error(std::string_view f, std::string_view msg, int c)

136 static validation_error missing_required(std::string_view field_name)
137 {
138 return validation_error(
139 field_name,
140 std::string("Required field '") + std::string(field_name) + "' is missing",
141 validation_codes::missing_required);
142 }
143
147 static validation_error type_mismatch(std::string_view field_name,
148 value_types expected,
149 value_types actual)
150 {
151 return validation_error(
152 field_name,
153 std::string("Field '") + std::string(field_name) + "' type mismatch: expected " +
154 std::string(get_string_from_type(expected)) + ", got " +
155 std::string(get_string_from_type(actual)),
156 validation_codes::type_mismatch);
157 }
158
162 template<typename T>
163 static validation_error out_of_range(std::string_view field_name,
164 T value, T min_val, T max_val)
165 {
166 return validation_error(
167 field_name,
168 std::string("Field '") + std::string(field_name) + "' value " +
169 std::to_string(value) + " is out of range [" +
170 std::to_string(min_val) + ", " + std::to_string(max_val) + "]",
171 validation_codes::out_of_range);
172 }
173
177 static validation_error invalid_length(std::string_view field_name,
178 size_t actual, size_t min_len, size_t max_len)
179 {
180 return validation_error(
181 field_name,
182 std::string("Field '") + std::string(field_name) + "' length " +
183 std::to_string(actual) + " is outside bounds [" +
184 std::to_string(min_len) + ", " + std::to_string(max_len) + "]",
185 validation_codes::invalid_length);
186 }
187
191 static validation_error pattern_mismatch(std::string_view field_name,
192 std::string_view pattern)
193 {
194 return validation_error(
195 field_name,
196 std::string("Field '") + std::string(field_name) +
197 "' does not match pattern '" + std::string(pattern) + "'",
198 validation_codes::pattern_mismatch);
199 }
200
204 static validation_error not_allowed(std::string_view field_name,
205 std::string_view value)
206 {
207 return validation_error(
208 field_name,
209 std::string("Field '") + std::string(field_name) + "' value '" +
210 std::string(value) + "' is not in allowed values",
211 validation_codes::not_in_allowed_values);
212 }
213
217 static validation_error custom_failed(std::string_view field_name,
218 std::string_view reason)
219 {
220 return validation_error(
221 field_name,
222 std::string("Field '") + std::string(field_name) +
223 "' custom validation failed: " + std::string(reason),
224 validation_codes::custom_validation_failed);
225 }
226
230 static validation_error nested_failed(std::string_view field_name,
231 const std::vector<validation_error>& nested_errors)
232 {
233 std::string msg = std::string("Field '") + std::string(field_name) +
234 "' nested validation failed with " +
235 std::to_string(nested_errors.size()) + " error(s)";
236 return validation_error(field_name, msg, validation_codes::nested_validation_failed);
237 }
