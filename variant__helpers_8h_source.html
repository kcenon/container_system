36 inline std::string json_escape(std::string_view input)
37 {
38 std::string result;
39 result.reserve(input.size() + input.size() / 8);
40
41 for (char c : input)
42 {
43 switch (c)
44 {
45 case '"':
46 result += "\\\"";
47 break;
48 case '\\':
49 result += "\\\\";
50 break;
51 case '\b':
52 result += "\\b";
53 break;
54 case '\f':
55 result += "\\f";
56 break;
57 case '\n':
58 result += "\\n";
59 break;
60 case '\r':
61 result += "\\r";
62 break;
63 case '\t':
64 result += "\\t";
65 break;
66 default:
67 if (static_cast<unsigned char>(c) < 0x20)
68 {
69 // Control character - use \uXXXX format
70 char buf[8];
71 std::snprintf(buf, sizeof(buf), "\\u%04x",
72 static_cast<unsigned char>(c));
73 result += buf;
74 }
75 else
76 {
77 result += c;
78 }
79 break;
80 }
81 }
82
83 return result;
84 }

91 inline std::string xml_encode(std::string_view input)
92 {
93 std::string result;
94 result.reserve(input.size() + input.size() / 8);
95
96 for (char c : input)
97 {
98 switch (c)
99 {
100 case '&':
101 result += "&amp;";
102 break;
103 case '<':
104 result += "&lt;";
105 break;
106 case '>':
107 result += "&gt;";
108 break;
109 case '"':
110 result += "&quot;";
111 break;
112 case '\'':
113 result += "&apos;";
114 break;
115 default:
116 if (static_cast<unsigned char>(c) < 0x20
117 && c != '\t' && c != '\n' && c != '\r')
118 {
119 // Control character - use numeric character reference
120 char buf[16];
121 std::snprintf(buf, sizeof(buf), "&#x%02x;",
122 static_cast<unsigned char>(c));
123 result += buf;
124 }
125 else
126 {
127 result += c;
128 }
129 break;
130 }
131 }
132
133 return result;
134 }
