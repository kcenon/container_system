175 constexpr std::string_view get_message(int code) noexcept

177 switch (code)
178 {
179 // Value operations (-400 to -409)

181 case type_mismatch: return "Type mismatch";
182 case value_out_of_range: return "Value out of range";
183 case invalid_value: return "Invalid value";
184 case key_already_exists: return "Key already exists";
185 case empty_key: return "Empty key";

187 // Validation (-410 to -419)
188 case schema_validation_failed: return "Schema validation failed";
189 case missing_required_field: return "Missing required field";
190 case constraint_violated: return "Constraint violated";
191 case type_constraint_violated: return "Type constraint violated";
192 case max_size_exceeded: return "Maximum size exceeded";
193
194 // Serialization (-420 to -429)
195 case serialization_failed: return "Serialization failed";
196 case deserialization_failed: return "Deserialization failed";
197 case invalid_format: return "Invalid format";
198 case version_mismatch: return "Version mismatch";
199 case corrupted_data: return "Corrupted data";
200 case header_parse_failed: return "Header parse failed";
201 case value_parse_failed: return "Value parse failed";
202 case encoding_error: return "Encoding error";
203
204 // Resource (-430 to -439)

206 case file_not_found: return "File not found";

231 constexpr std::string_view get_category(int code) noexcept
232 {
233 if (code <= -400 && code > -410)
234 {
235 return "value_operation";
236 }
237 if (code <= -410 && code > -420)
238 {
239 return "validation";
240 }
241 if (code <= -420 && code > -430)
242 {
243 return "serialization";
244 }
245 if (code <= -430 && code > -440)
246 {
247 return "resource";
248 }
249 if (code <= -440 && code > -450)
250 {
251 return "thread_safety";
252 }
253 return "unknown";
254 }
