92std::string message_container::serialize() const
93{
94#if HAS_NLOHMANN_JSON == 1
95 nlohmann::json json_obj;
96
97 // Serialize header
98 nlohmann::json header;
99 header["source_id"] = source_id_;
100 header["source_sub_id"] = source_sub_id_;
101 header["target_id"] = target_id_;
102 header["target_sub_id"] = target_sub_id_;
103 header["message_type"] = message_type_;
104 header["version"] = version_;
105
106 json_obj["header"] = header;
107
108 // Serialize payload
109 json_obj["payload"] = payload_.serialize();
110
111 return json_obj.dump();
112#else
113 // Fallback: simple JSON-like format
114 std::ostringstream oss;
115 oss << "{"
116 << "\"header\":{"
117 << "\"source_id\":\"" << source_id_ << "\","
118 << "\"source_sub_id\":\"" << source_sub_id_ << "\","
119 << "\"target_id\":\"" << target_id_ << "\","
120 << "\"target_sub_id\":\"" << target_sub_id_ << "\","
121 << "\"message_type\":\"" << message_type_ << "\","
122 << "\"version\":\"" << version_ << "\""
123 << "},"
124 << "\"payload\":" << payload_.serialize()
125 << "}";
126 return oss.str();
127#endif
128}

130std::vector<uint8_t> message_container::serialize_binary() const
131{
132 // Binary format: [header_size][header_data][payload_data]
133 std::vector<uint8_t> result;
134
135 // Serialize header to JSON first (can be optimized to binary format)
136 std::string header_json = "{"
137 "\"source_id\":\"" + source_id_ + "\","
138 "\"source_sub_id\":\"" + source_sub_id_ + "\","
139 "\"target_id\":\"" + target_id_ + "\","
140 "\"target_sub_id\":\"" + target_sub_id_ + "\","
141 "\"message_type\":\"" + message_type_ + "\","
142 "\"version\":\"" + version_ + "\""
143 "}";
144
145 // Write header size (4 bytes)
146 uint32_t header_size = static_cast<uint32_t>(header_json.size());
147 result.push_back(static_cast<uint8_t>(header_size & 0xFF));
148 result.push_back(static_cast<uint8_t>((header_size >> 8) & 0xFF));
149 result.push_back(static_cast<uint8_t>((header_size >> 16) & 0xFF));
150 result.push_back(static_cast<uint8_t>((header_size >> 24) & 0xFF));
151
152 // Write header data
153 result.insert(result.end(), header_json.begin(), header_json.end());
154
155 // Write payload data
156 auto payload_data = payload_.serialize_binary();
157 result.insert(result.end(), payload_data.begin(), payload_data.end());
158
159 return result;
160}

162std::unique_ptr<message_container> message_container::deserialize(std::string_view json_data)
163{
164#if HAS_NLOHMANN_JSON == 1
165 auto json_obj = nlohmann::json::parse(json_data);
166
167 auto container = std::make_unique<message_container>();
168
169 // Deserialize header
170 if (json_obj.contains("header")) {
171 auto header = json_obj["header"];
172 if (header.contains("source_id")) container->source_id_ = header["source_id"];
173 if (header.contains("source_sub_id")) container->source_sub_id_ = header["source_sub_id"];
174 if (header.contains("target_id")) container->target_id_ = header["target_id"];
175 if (header.contains("target_sub_id")) container->target_sub_id_ = header["target_sub_id"];
176 if (header.contains("message_type")) container->message_type_ = header["message_type"];
177 if (header.contains("version")) container->version_ = header["version"];
178 }
179
180 // Deserialize payload in place into payload_
181 // (value_store is non-copyable/non-movable, so an in-place loader is required)
182 if (json_obj.contains("payload")) {
183 // serialize() stores the payload as a JSON string; also accept the
184 // object form emitted by the fallback serializer.
185 const auto& payload = json_obj["payload"];
186 std::string payload_str = payload.is_string()
187 ? payload.get<std::string>() : payload.dump();
188 value_store::deserialize_into(container->payload_, payload_str);
189 }
190
191 return container;
192#else
193 throw std::runtime_error("JSON library not available for deserialization");
194#endif
195}

197std::unique_ptr<message_container> message_container::deserialize_binary(const std::vector<uint8_t>& binary_data)
198{
199 if (binary_data.size() < 4) {
200 throw std::runtime_error("Binary data too small to contain header size");
201 }
202
203 // Read header size (4 bytes)
204 uint32_t header_size = static_cast<uint32_t>(binary_data[0])
205 | (static_cast<uint32_t>(binary_data[1]) << 8)
206 | (static_cast<uint32_t>(binary_data[2]) << 16)
207 | (static_cast<uint32_t>(binary_data[3]) << 24);
208
209 if (binary_data.size() < 4 + header_size) {
210 throw std::runtime_error("Binary data too small to contain header");
211 }
212
213 // Read header data
214 std::string header_json(binary_data.begin() + 4, binary_data.begin() + 4 + header_size);
215
216 auto container = std::make_unique<message_container>();
217
218#if HAS_NLOHMANN_JSON == 1
219 auto header = nlohmann::json::parse(header_json);
220 if (header.contains("source_id")) container->source_id_ = header["source_id"];
221 if (header.contains("source_sub_id")) container->source_sub_id_ = header["source_sub_id"];
222 if (header.contains("target_id")) container->target_id_ = header["target_id"];
223 if (header.contains("target_sub_id")) container->target_sub_id_ = header["target_sub_id"];
224 if (header.contains("message_type")) container->message_type_ = header["message_type"];
225 if (header.contains("version")) container->version_ = header["version"];
226#else
227 // Simple parsing fallback (not recommended for production)
228 // This is a simplified version - proper JSON parsing would be needed
229#endif
230
231 // Read payload data and deserialize it in place into payload_
232 // (value_store is non-copyable/non-movable, so an in-place loader is required)
233 if (binary_data.size() > 4 + header_size) {
234 std::vector<uint8_t> payload_data(binary_data.begin() + 4 + header_size,
235 binary_data.end());
236 value_store::deserialize_binary_into(container->payload_, payload_data);
237 }
238
239 return container;
240}
