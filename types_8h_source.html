77 struct optimized_value
78 {
79 std::string name;
80 value_types type;

83 optimized_value()
84 : name("")
85 , type(value_types::null_value)
86 , data(std::monostate{})

88
89 optimized_value(const std::string& n, value_types t)
90 : name(n)
91 , type(t)
92 , data(std::monostate{})
93 {}
94
95 size_t memory_footprint() const
96 {
97 size_t base = sizeof(optimized_value);
98 base += name.capacity();
99 if (std::holds_alternative<std::string>(data)) {
100 base += std::get<std::string>(data).capacity();
101 } else if (std::holds_alternative<std::vector<uint8_t>>(data)) {
102 base += std::get<std::vector<uint8_t>>(data).capacity();
103 }
104 return base;
105 }
106
107 bool is_stack_allocated() const
108 {
109 return type != value_types::string_value &&
110 type != value_types::bytes_value &&
111 type != value_types::container_value;
112 }
113 };

118 struct pool_stats
119 {
120 size_t hits{0}; // Pool allocations satisfied
121 size_t misses{0}; // Heap allocations (pool bypassed)
122 size_t small_pool_allocs{0}; // Small pool (<=64 bytes) allocations
123 size_t medium_pool_allocs{0}; // Medium pool (<=256 bytes) allocations
124 size_t deallocations{0}; // Total deallocations
125 size_t available{0}; // Free blocks available
126
127 pool_stats() = default;
128
129 pool_stats(size_t h, size_t m, size_t a)
130 : hits(h), misses(m), available(a)
131 {}
132
133 pool_stats(size_t h, size_t m, size_t sm, size_t med, size_t deallocs, size_t avail)
134 : hits(h), misses(m), small_pool_allocs(sm), medium_pool_allocs(med)
135 , deallocations(deallocs), available(avail)
136 {}
137
138 [[nodiscard]] double hit_rate() const noexcept {
139 auto total = hits + misses;
140 return total > 0 ? static_cast<double>(hits) / static_cast<double>(total) : 0.0;
141 }
