131 void write_nil()

139 void write_bool(bool value)

141 buffer_.push_back(value ? msgpack_format::TRUE : msgpack_format::FALSE);

147 void write_int(int64_t value)
148 {
149 if (value >= 0)
150 {
151 write_uint(static_cast<uint64_t>(value));
152 }
153 else if (value >= -32)
154 {
155 // Negative fixint
156 buffer_.push_back(static_cast<uint8_t>(value));
157 }
158 else if (value >= INT8_MIN)
159 {
160 buffer_.push_back(msgpack_format::INT8);
161 buffer_.push_back(static_cast<uint8_t>(value));
162 }
163 else if (value >= INT16_MIN)
164 {
165 buffer_.push_back(msgpack_format::INT16);
166 write_be16(static_cast<uint16_t>(value));
167 }
168 else if (value >= INT32_MIN)
169 {
170 buffer_.push_back(msgpack_format::INT32);
171 write_be32(static_cast<uint32_t>(value));
172 }
173 else
174 {
175 buffer_.push_back(msgpack_format::INT64);
176 write_be64(static_cast<uint64_t>(value));
177 }
178 }
179
183 void write_uint(uint64_t value)
184 {
185 if (value <= msgpack_format::POSITIVE_FIXINT_MAX)
186 {
187 // Positive fixint
188 buffer_.push_back(static_cast<uint8_t>(value));
189 }
190 else if (value <= UINT8_MAX)
191 {
192 buffer_.push_back(msgpack_format::UINT8);
193 buffer_.push_back(static_cast<uint8_t>(value));
194 }
195 else if (value <= UINT16_MAX)
196 {
197 buffer_.push_back(msgpack_format::UINT16);
198 write_be16(static_cast<uint16_t>(value));
199 }
200 else if (value <= UINT32_MAX)
201 {
202 buffer_.push_back(msgpack_format::UINT32);
203 write_be32(static_cast<uint32_t>(value));
204 }
205 else
206 {
207 buffer_.push_back(msgpack_format::UINT64);
208 write_be64(value);
209 }
210 }

237 void write_string(std::string_view value)
238 {

241 {
242 buffer_.push_back(static_cast<uint8_t>(msgpack_format::FIXSTR_PREFIX | len));

244 else if (len <= UINT8_MAX)

251 buffer_.push_back(msgpack_format::STR16);

253 }
254 else
255 {
256 buffer_.push_back(msgpack_format::STR32);
257 write_be32(static_cast<uint32_t>(len));
258 }
259 buffer_.insert(buffer_.end(), value.begin(), value.end());
260 }

354 {
355 buffer_.push_back(static_cast<uint8_t>(value >> 24));
356 buffer_.push_back(static_cast<uint8_t>(value >> 16));
357 buffer_.push_back(static_cast<uint8_t>(value >> 8));
358 buffer_.push_back(static_cast<uint8_t>(value));
359 }

362 {
363 buffer_.push_back(static_cast<uint8_t>(value >> 56));
364 buffer_.push_back(static_cast<uint8_t>(value >> 48));
365 buffer_.push_back(static_cast<uint8_t>(value >> 40));
366 buffer_.push_back(static_cast<uint8_t>(value >> 32));
367 buffer_.push_back(static_cast<uint8_t>(value >> 24));
368 buffer_.push_back(static_cast<uint8_t>(value >> 16));
369 buffer_.push_back(static_cast<uint8_t>(value >> 8));
370 buffer_.push_back(static_cast<uint8_t>(value));
371 }

395 msgpack_type peek_type() const
396 {
397 if (offset_ >= size_)
398 {
399 return msgpack_type::unknown;
400 }
401
402 uint8_t byte = data_[offset_];
403
404 // Positive fixint
405 if (byte <= msgpack_format::POSITIVE_FIXINT_MAX)
406 {
407 return msgpack_type::positive_int;
408 }
409
410 // Fixmap
411 if ((byte & 0xf0) == msgpack_format::FIXMAP_PREFIX)
412 {
413 return msgpack_type::map;
414 }
415
416 // Fixarray
417 if ((byte & 0xf0) == msgpack_format::FIXARRAY_PREFIX)
418 {
419 return msgpack_type::array;
420 }
421
422 // Fixstr
423 if ((byte & 0xe0) == msgpack_format::FIXSTR_PREFIX)
424 {
425 return msgpack_type::str;
426 }
427
428 // Negative fixint
429 if (byte >= msgpack_format::NEGATIVE_FIXINT_PREFIX)
430 {
431 return msgpack_type::negative_int;
432 }
433
434 switch (byte)
435 {
436 case msgpack_format::NIL:
437 return msgpack_type::nil;
438 case msgpack_format::FALSE:
439 case msgpack_format::TRUE:
440 return msgpack_type::boolean;
441 case msgpack_format::BIN8:
442 case msgpack_format::BIN16:
443 case msgpack_format::BIN32:
444 return msgpack_type::bin;
445 case msgpack_format::FLOAT32:
446 return msgpack_type::float32;
447 case msgpack_format::FLOAT64:
448 return msgpack_type::float64;
449 case msgpack_format::UINT8:
450 case msgpack_format::UINT16:
451 case msgpack_format::UINT32:
452 case msgpack_format::UINT64:
453 return msgpack_type::positive_int;
454 case msgpack_format::INT8:
455 case msgpack_format::INT16:
456 case msgpack_format::INT32:
457 case msgpack_format::INT64:
458 return msgpack_type::negative_int;
459 case msgpack_format::STR8:
460 case msgpack_format::STR16:
461 case msgpack_format::STR32:
462 return msgpack_type::str;
463 case msgpack_format::ARRAY16:
464 case msgpack_format::ARRAY32:
465 return msgpack_type::array;
466 case msgpack_format::MAP16:
467 case msgpack_format::MAP32:
468 return msgpack_type::map;
469 default:
470 return msgpack_type::unknown;
471 }
472 }

491 std::optional<bool> read_bool()

498 uint8_t byte = data_[offset_];

515 std::optional<int64_t> read_int()
516 {
517 if (offset_ >= size_)
518 {
519 return std::nullopt;
520 }
521
522 uint8_t byte = data_[offset_];
523
524 // Positive fixint
525 if (byte <= msgpack_format::POSITIVE_FIXINT_MAX)
526 {
527 ++offset_;
528 return static_cast<int64_t>(byte);
529 }
530
531 // Negative fixint
532 if (byte >= msgpack_format::NEGATIVE_FIXINT_PREFIX)
533 {
534 ++offset_;
535 return static_cast<int64_t>(static_cast<int8_t>(byte));
536 }
537
538 switch (byte)
539 {
540 case msgpack_format::INT8:
541 if (offset_ + 2 > size_)
542 {
543 return std::nullopt;
544 }
545 ++offset_;
546 return static_cast<int64_t>(static_cast<int8_t>(data_[offset_++]));
547
548 case msgpack_format::INT16:
549 if (offset_ + 3 > size_)
550 {
551 return std::nullopt;
552 }
553 ++offset_;
554 return static_cast<int64_t>(static_cast<int16_t>(read_be16()));
555
556 case msgpack_format::INT32:
557 if (offset_ + 5 > size_)
558 {
559 return std::nullopt;
560 }
561 ++offset_;
562 return static_cast<int64_t>(static_cast<int32_t>(read_be32()));
563
564 case msgpack_format::INT64:
565 if (offset_ + 9 > size_)
566 {
567 return std::nullopt;
568 }
569 ++offset_;
570 return static_cast<int64_t>(read_be64());
571
572 case msgpack_format::UINT8:
573 if (offset_ + 2 > size_)
574 {
575 return std::nullopt;
576 }
577 ++offset_;
578 return static_cast<int64_t>(data_[offset_++]);
579
580 case msgpack_format::UINT16:
581 if (offset_ + 3 > size_)
582 {
583 return std::nullopt;
584 }
585 ++offset_;
586 return static_cast<int64_t>(read_be16());
587
588 case msgpack_format::UINT32:
589 if (offset_ + 5 > size_)
590 {
591 return std::nullopt;
592 }
593 ++offset_;
594 return static_cast<int64_t>(read_be32());
595
596 case msgpack_format::UINT64:
597 if (offset_ + 9 > size_)
598 {
599 return std::nullopt;
600 }
601 ++offset_;
602 {
603 uint64_t val = read_be64();
604 if (val > static_cast<uint64_t>(INT64_MAX))
605 {
606 return std::nullopt; // Overflow
607 }
608 return static_cast<int64_t>(val);
609 }
610
611 default:
612 return std::nullopt;
613 }
614 }

735 std::optional<std::string> read_string()
736 {
737 if (offset_ >= size_)
738 {
739 return std::nullopt;
740 }
741
742 uint8_t byte = data_[offset_];
743 size_t len = 0;
744
745 // Fixstr
746 if ((byte & 0xe0) == msgpack_format::FIXSTR_PREFIX)
747 {
748 len = byte & 0x1f;
749 ++offset_;
750 }
751 else
752 {
753 switch (byte)
754 {
755 case msgpack_format::STR8:
756 if (offset_ + 2 > size_) return std::nullopt;
757 ++offset_;

759 break;
760
761 case msgpack_format::STR16:
762 if (offset_ + 3 > size_) return std::nullopt;
763 ++offset_;
764 len = read_be16();
765 break;
766
767 case msgpack_format::STR32:
768 if (offset_ + 5 > size_) return std::nullopt;
769 ++offset_;
770 len = read_be32();
771 break;
772
773 default:
774 return std::nullopt;
775 }
776 }
777
778 if (offset_ + len > size_)
779 {
780 return std::nullopt;
781 }
782
783 std::string result(reinterpret_cast<const char*>(data_ + offset_), len);
784 offset_ += len;
785 return result;

923 uint16_t read_be16()
924 {
925 uint16_t result = (static_cast<uint16_t>(data_[offset_]) << 8)
926 | static_cast<uint16_t>(data_[offset_ + 1]);
927 offset_ += 2;
928 return result;

931 uint32_t read_be32()
932 {
933 uint32_t result = (static_cast<uint32_t>(data_[offset_]) << 24)
934 | (static_cast<uint32_t>(data_[offset_ + 1]) << 16)
935 | (static_cast<uint32_t>(data_[offset_ + 2]) << 8)
936 | static_cast<uint32_t>(data_[offset_ + 3]);
937 offset_ += 4;
938 return result;
939 }

944 | (static_cast<uint64_t>(data_[offset_ + 1]) << 48)
945 | (static_cast<uint64_t>(data_[offset_ + 2]) << 40)
946 | (static_cast<uint64_t>(data_[offset_ + 3]) << 32)
947 | (static_cast<uint64_t>(data_[offset_ + 4]) << 24)
948 | (static_cast<uint64_t>(data_[offset_ + 5]) << 16)
949 | (static_cast<uint64_t>(data_[offset_ + 6]) << 8)
950 | static_cast<uint64_t>(data_[offset_ + 7]);
