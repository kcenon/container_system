144 struct latency_histogram
145 {
146 static constexpr size_t kReservoirSize = 1024;
147
148 std::atomic<uint64_t> sample_count{0};
149 std::atomic<uint64_t> max_ns{0};
150 std::atomic<uint64_t> min_ns{UINT64_MAX};
151 std::atomic<uint64_t> sum_ns{0};
152
153 // Reservoir for percentile calculation
154 std::array<std::atomic<uint64_t>, kReservoirSize> reservoir{};
155 std::atomic<size_t> reservoir_count{0};
156
157 latency_histogram() = default;
158 ~latency_histogram() = default;
159
160 latency_histogram(const latency_histogram& other) noexcept
161 : sample_count(other.sample_count.load(std::memory_order_relaxed))
162 , max_ns(other.max_ns.load(std::memory_order_relaxed))
163 , min_ns(other.min_ns.load(std::memory_order_relaxed))
164 , sum_ns(other.sum_ns.load(std::memory_order_relaxed))
165 , reservoir_count(other.reservoir_count.load(std::memory_order_relaxed))
166 {
167 size_t count = reservoir_count.load(std::memory_order_relaxed);
168 for (size_t i = 0; i < count && i < kReservoirSize; ++i)
169 {
170 reservoir[i].store(other.reservoir[i].load(std::memory_order_relaxed),
171 std::memory_order_relaxed);
172 }
173 }
174
175 latency_histogram& operator=(const latency_histogram& other) noexcept
176 {
177 if (this != &other)
178 {
179 sample_count.store(other.sample_count.load(std::memory_order_relaxed), std::memory_order_relaxed);
180 max_ns.store(other.max_ns.load(std::memory_order_relaxed), std::memory_order_relaxed);
181 min_ns.store(other.min_ns.load(std::memory_order_relaxed), std::memory_order_relaxed);
182 sum_ns.store(other.sum_ns.load(std::memory_order_relaxed), std::memory_order_relaxed);
183 reservoir_count.store(other.reservoir_count.load(std::memory_order_relaxed), std::memory_order_relaxed);
184
185 size_t count = reservoir_count.load(std::memory_order_relaxed);
186 for (size_t i = 0; i < count && i < kReservoirSize; ++i)
187 {
188 reservoir[i].store(other.reservoir[i].load(std::memory_order_relaxed),
189 std::memory_order_relaxed);
190 }
191 }
192 return *this;

199 void record(uint64_t latency_ns) noexcept
200 {
201 // Update sample count
202 uint64_t count = sample_count.fetch_add(1, std::memory_order_relaxed) + 1;
203
204 // Update max
205 uint64_t current_max = max_ns.load(std::memory_order_relaxed);
206 while (latency_ns > current_max &&
207 !max_ns.compare_exchange_weak(current_max, latency_ns,
208 std::memory_order_relaxed))
209 {
210 }
211
212 // Update min
213 uint64_t current_min = min_ns.load(std::memory_order_relaxed);
214 while (latency_ns < current_min &&
215 !min_ns.compare_exchange_weak(current_min, latency_ns,
216 std::memory_order_relaxed))
217 {
218 }
219
220 // Update sum
221 sum_ns.fetch_add(latency_ns, std::memory_order_relaxed);
222
223 // Reservoir sampling
224 size_t idx = reservoir_count.load(std::memory_order_relaxed);
225 if (idx < kReservoirSize)
226 {
227 // Fill the reservoir initially
228 reservoir[idx].store(latency_ns, std::memory_order_relaxed);
229 reservoir_count.fetch_add(1, std::memory_order_relaxed);
230 }
231 else
232 {
233 // Probabilistic replacement for uniform sampling
234 // Use simple hash of count as pseudo-random
235 size_t replace_idx = (count * 2654435769ULL) % kReservoirSize;
236 if ((count * 2654435769ULL >> 32) % count < kReservoirSize)
237 {
238 reservoir[replace_idx].store(latency_ns, std::memory_order_relaxed);
239 }
240 }
241 }
242
248 [[nodiscard]] uint64_t get_percentile(double percentile) const noexcept
249 {
250 size_t count = reservoir_count.load(std::memory_order_relaxed);
251 if (count == 0)
252 {
253 return 0;
254 }
255
256 // Copy reservoir values to local array for sorting
257 std::array<uint64_t, kReservoirSize> values{};
258 for (size_t i = 0; i < count && i < kReservoirSize; ++i)
259 {
260 values[i] = reservoir[i].load(std::memory_order_relaxed);
261 }
262
263 // Sort the values
264 std::sort(values.begin(), values.begin() + count);
265
266 // Calculate percentile index
267 size_t idx = static_cast<size_t>(percentile * static_cast<double>(count - 1));
268 if (idx >= count)
269 {
270 idx = count - 1;
271 }
272
273 return values[idx];
274 }
275
279 [[nodiscard]] uint64_t p50() const noexcept { return get_percentile(0.50); }
280
284 [[nodiscard]] uint64_t p95() const noexcept { return get_percentile(0.95); }
285
289 [[nodiscard]] uint64_t p99() const noexcept { return get_percentile(0.99); }
290
294 [[nodiscard]] uint64_t p999() const noexcept { return get_percentile(0.999); }
295
299 [[nodiscard]] double avg() const noexcept
300 {
301 uint64_t count = sample_count.load(std::memory_order_relaxed);
302 if (count == 0)
303 {
304 return 0.0;
305 }
306 return static_cast<double>(sum_ns.load(std::memory_order_relaxed)) /
307 static_cast<double>(count);
308 }
309
313 void reset() noexcept
314 {
315 sample_count.store(0, std::memory_order_relaxed);
316 max_ns.store(0, std::memory_order_relaxed);
317 min_ns.store(UINT64_MAX, std::memory_order_relaxed);
318 sum_ns.store(0, std::memory_order_relaxed);
319 reservoir_count.store(0, std::memory_order_relaxed);
320 for (auto& val : reservoir)
321 {
322 val.store(0, std::memory_order_relaxed);
323 }
324 }

551 {
552 public:
558 explicit scoped_timer(latency_histogram& histogram,
559 std::atomic<uint64_t>* timing_total = nullptr) noexcept
560 : histogram_(histogram)
561 , timing_total_(timing_total)
562 , start_(std::chrono::steady_clock::now())

566
570 scoped_timer() noexcept
571 : histogram_(dummy_histogram_)
572 , timing_total_(nullptr)
573 , enabled_(false)
574 {
575 }
576
577 ~scoped_timer() noexcept
578 {
579 if (enabled_)
580 {
581 auto end = std::chrono::steady_clock::now();
582 auto duration = std::chrono::duration_cast<std::chrono::nanoseconds>(
583 end - start_).count();
584 auto latency = static_cast<uint64_t>(duration);
585
586 histogram_.record(latency);
587 if (timing_total_)
588 {
589 timing_total_->fetch_add(latency, std::memory_order_relaxed);
590 }
591 }
592 }

602 std::atomic<uint64_t>* timing_total_;
603 std::chrono::steady_clock::time_point start_;
