113 {"15", value_types::array_value}
114}};

121constexpr value_types get_type_from_string(std::string_view str) noexcept {
122 for (const auto& [key, type] : type_map) {
123 if (key == str) return type;
124 }
125 return value_types::null_value;
126}
127
133constexpr std::string_view get_string_from_type(value_types type) noexcept {
134 for (const auto& [key, val] : type_map) {
135 if (val == type) return key;
136 }
137 return "0";
138}

145inline value_types convert_value_type(const std::string& target) {
146 for (const auto& [key, type] : type_map) {
147 if (key == target) return type;
148 }
149 return value_types::null_value;
150}

208struct optimized_value {
209 std::string name;
210 value_types type;
211 value_variant data;
212
214 optimized_value()
215 : name("")
216 , type(value_types::null_value)
217 , data(std::monostate{})
218 {}
219
225 optimized_value(const std::string& n, value_types t)
226 : name(n)
227 , type(t)
228 , data(std::monostate{})
229 {}
230
235 size_t memory_footprint() const {
236 size_t base = sizeof(optimized_value);
237 base += name.capacity();
238 if (std::holds_alternative<std::string>(data)) {
239 base += std::get<std::string>(data).capacity();
240 } else if (std::holds_alternative<std::vector<uint8_t>>(data)) {
241 base += std::get<std::vector<uint8_t>>(data).capacity();
242 }
243 return base;
244 }
245
250 bool is_stack_allocated() const {
251 return type != value_types::string_value &&
252 type != value_types::bytes_value &&
253 type != value_types::container_value;
254 }
255};

297inline std::string json_escape(std::string_view input) {
298 std::string result;
299 result.reserve(input.size() + input.size() / 8);
300
301 for (char c : input) {
302 switch (c) {
303 case '"':
304 result += "\\\"";
305 break;
306 case '\\':
307 result += "\\\\";
308 break;
309 case '\b':
310 result += "\\b";
311 break;
312 case '\f':
313 result += "\\f";
314 break;
315 case '\n':
316 result += "\\n";
317 break;
318 case '\r':
319 result += "\\r";
320 break;
321 case '\t':
322 result += "\\t";
323 break;
324 default:
325 if (static_cast<unsigned char>(c) < 0x20) {
326 char buf[8];
327 std::snprintf(buf, sizeof(buf), "\\u%04x",
328 static_cast<unsigned char>(c));
329 result += buf;
330 } else {
331 result += c;
332 }
333 break;
334 }
335 }
336
337 return result;
338}
339
345inline std::string xml_encode(std::string_view input) {
346 std::string result;
347 result.reserve(input.size() + input.size() / 8);
348
349 for (char c : input) {
350 switch (c) {
351 case '&':
352 result += "&amp;";
353 break;
354 case '<':
355 result += "&lt;";
356 break;
357 case '>':
358 result += "&gt;";
359 break;
360 case '"':
361 result += "&quot;";
362 break;
363 case '\'':
364 result += "&apos;";
365 break;
366 default:
367 if (static_cast<unsigned char>(c) < 0x20
368 && c != '\t' && c != '\n' && c != '\r') {
369 char buf[16];
370 std::snprintf(buf, sizeof(buf), "&#x%02x;",
371 static_cast<unsigned char>(c));
372 result += buf;
373 } else {
374 result += c;
375 }
376 break;
377 }
378 }
379
380 return result;
381}

386inline std::string to_string(const value_variant& var, value_types type) {
387 switch (type) {
388 case value_types::null_value:
389 return "";
390 case value_types::bool_value:
391 return std::get<bool>(var) ? "true" : "false";
392 case value_types::short_value:
393 return std::to_string(std::get<short>(var));
394 case value_types::ushort_value:
395 return std::to_string(std::get<unsigned short>(var));
396 case value_types::int_value:
397 return std::to_string(std::get<int>(var));
398 case value_types::uint_value:
399 return std::to_string(std::get<unsigned int>(var));
400 case value_types::long_value:
401 return std::to_string(std::get<long>(var));
402 case value_types::ulong_value:
403 return std::to_string(std::get<unsigned long>(var));
404 case value_types::llong_value:
405 return std::to_string(std::get<long long>(var));
406 case value_types::ullong_value:
407 return std::to_string(std::get<unsigned long long>(var));
408 case value_types::float_value:
409 return std::to_string(std::get<float>(var));
410 case value_types::double_value:
411 return std::to_string(std::get<double>(var));
412 case value_types::string_value:
413 return std::get<std::string>(var);
414 default:
415 return "";
416 }
