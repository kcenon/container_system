27// Reconstruct a single value from its JSON object representation, mirroring
28// the format produced by value::to_json():
29// {"name":"<name>","type":<int 0-15>,"value":<typed-json>}
30value value_from_json(const std::string& key, const nlohmann::json& node) {
31 if (!node.is_object() || !node.contains("type") || !node.contains("value")) {
32 throw std::runtime_error(
33 "value_store::deserialize() - malformed value object for key: " + key);
34 }
35
36 const std::string name = node.contains("name") && node["name"].is_string()
37 ? node["name"].get<std::string>()
38 : key;
39 const int type_int = node["type"].get<int>();
40 const nlohmann::json& v = node["value"];
41
42 switch (static_cast<value_types>(type_int)) {
43 case value_types::null_value:
44 return value(name);
45 case value_types::bool_value:
46 return value(name, v.get<bool>());
47 case value_types::short_value:
48 return value(name, static_cast<int16_t>(v.get<int>()));
49 case value_types::ushort_value:
50 return value(name, static_cast<uint16_t>(v.get<unsigned int>()));
51 case value_types::int_value:
52 return value(name, v.get<int32_t>());
53 case value_types::uint_value:
54 return value(name, v.get<uint32_t>());
55 case value_types::long_value:
56 return value(name, v.get<long>());
57 case value_types::llong_value:
58 return value(name, v.get<long long>());
59 case value_types::ulong_value:
60 return value(name, v.get<unsigned long>());
61 case value_types::ullong_value:
62 return value(name, v.get<unsigned long long>());
63 case value_types::float_value:
64 return value(name, v.get<float>());
65 case value_types::double_value:
66 return value(name, v.get<double>());
67 case value_types::string_value:
68 return value(name, v.get<std::string>());
69 case value_types::bytes_value: {
70 // to_json() renders bytes as a lowercase hex string
71 const std::string hex = v.get<std::string>();
72 if (hex.size() % 2 != 0) {
73 throw std::runtime_error(
74 "value_store::deserialize() - odd-length hex for key: " + key);
75 }
76 std::vector<uint8_t> bytes;
77 bytes.reserve(hex.size() / 2);
78 for (size_t i = 0; i < hex.size(); i += 2) {
79 bytes.push_back(static_cast<uint8_t>(
80 std::stoul(hex.substr(i, 2), nullptr, 16)));
81 }
82 return value(name, std::move(bytes));
83 }
84 case value_types::container_value:
85 case value_types::array_value:
86 default:
87 throw std::runtime_error(
88 "value_store::deserialize() - unsupported value type for key '" + key
89 + "': " + std::to_string(type_int));
90 }
91}

153std::string value_store::serialize_impl() const {
154 read_count_.fetch_add(1, std::memory_order_relaxed);
155
156 std::string result = "{";
157 bool first = true;
158
159 for (const auto& [key, val] : values_) {
160 if (!first) {
161 result += ",";
162 }
163 first = false;
164
165 // Escape key for JSON
166 result += "\"";
167 for (char c : key) {
168 switch (c) {
169 case '"': result += "\\\""; break;
170 case '\\': result += "\\\\"; break;
171 case '\n': result += "\\n"; break;
172 case '\r': result += "\\r"; break;
173 case '\t': result += "\\t"; break;
174 default: result += c;
175 }
176 }
177 result += "\":";
178 result += val.to_json();
179 }
180
181 result += "}";
182 return result;
183}

191std::vector<uint8_t> value_store::serialize_binary_impl() const {
192 read_count_.fetch_add(1, std::memory_order_relaxed);
193
194 std::vector<uint8_t> result;
195
196 // Version byte for future compatibility
197 constexpr uint8_t version = 1;
198 result.push_back(version);
199
200 // Header: number of entries (4 bytes)
201 uint32_t count = static_cast<uint32_t>(values_.size());
202 result.insert(result.end(),
203 reinterpret_cast<const uint8_t*>(&count),
204 reinterpret_cast<const uint8_t*>(&count) + sizeof(count));
205
206 // Serialize each key-value pair
207 for (const auto& [key, val] : values_) {
208 // Key length and key
209 uint32_t key_len = static_cast<uint32_t>(key.size());
210 result.insert(result.end(),
211 reinterpret_cast<const uint8_t*>(&key_len),
212 reinterpret_cast<const uint8_t*>(&key_len) + sizeof(key_len));
213 result.insert(result.end(), key.begin(), key.end());
214
215 // Value serialization
216 auto value_data = val.serialize();
217 uint32_t value_len = static_cast<uint32_t>(value_data.size());
218 result.insert(result.end(),
219 reinterpret_cast<const uint8_t*>(&value_len),
220 reinterpret_cast<const uint8_t*>(&value_len) + sizeof(value_len));
221 result.insert(result.end(), value_data.begin(), value_data.end());
222 }
223
224 return result;
225}

239void value_store::deserialize_into(value_store& store, std::string_view json_data) {
240#if HAS_NLOHMANN_JSON == 1
241 auto json_obj = nlohmann::json::parse(json_data);
242 if (!json_obj.is_object()) {
243 throw std::runtime_error(
244 "value_store::deserialize() - top-level JSON value must be an object");
245 }
246
247 // Parse into a temporary map so a malformed input leaves the target unmodified
248 std::unordered_map<std::string, value> parsed;
249 for (auto it = json_obj.begin(); it != json_obj.end(); ++it) {
250 parsed.emplace(it.key(), value_from_json(it.key(), it.value()));
251 }
252
253 // Commit parsed entries atomically once the whole document is validated
254 std::unique_lock lock(store.mutex_);
255 store.values_ = std::move(parsed);
256#else
257 (void)store;
258 (void)json_data;
259 throw std::runtime_error(
260 "value_store::deserialize() requires JSON parser - use deserialize_binary() instead");
261#endif
262}

270void value_store::deserialize_binary_into(value_store& store,
271 const std::vector<uint8_t>& binary_data) {
272 if (binary_data.size() < 1 + sizeof(uint32_t)) {
273 throw std::runtime_error("value_store::deserialize_binary() - invalid data: too small");
274 }
275
276 size_t offset = 0;
277
278 // Read version byte
279 uint8_t version = binary_data[offset++];
280 if (version != 1) {
281 throw std::runtime_error("value_store::deserialize_binary() - unsupported version: "
282 + std::to_string(version));
283 }
284
285 // Read number of entries
286 uint32_t count;
287 std::memcpy(&count, binary_data.data() + offset, sizeof(count));
288 offset += sizeof(count);
289
290 // Parse into a temporary map so a malformed input leaves the target unmodified
291 std::unordered_map<std::string, value> parsed;
292
293 // Read each key-value pair
294 for (uint32_t i = 0; i < count; ++i) {
295 if (offset + sizeof(uint32_t) > binary_data.size()) {
296 throw std::runtime_error("value_store::deserialize_binary() - truncated data at entry "
297 + std::to_string(i));
298 }
299
300 // Read key length
301 uint32_t key_len;
302 std::memcpy(&key_len, binary_data.data() + offset, sizeof(key_len));
303 offset += sizeof(key_len);
304
305 if (offset + key_len + sizeof(uint32_t) > binary_data.size()) {
306 throw std::runtime_error("value_store::deserialize_binary() - truncated key data");
307 }
308
309 // Read key
310 std::string key(binary_data.begin() + offset,
311 binary_data.begin() + offset + key_len);
312 offset += key_len;
313
314 // Read value length
315 uint32_t value_len;
316 std::memcpy(&value_len, binary_data.data() + offset, sizeof(value_len));
317 offset += sizeof(value_len);
318
319 if (offset + value_len > binary_data.size()) {
320 throw std::runtime_error("value_store::deserialize_binary() - truncated value data");
321 }
322
323 // Deserialize value
324 std::vector<uint8_t> value_data(binary_data.begin() + offset,
325 binary_data.begin() + offset + value_len);
326 offset += value_len;
327
328 auto value_opt = value::deserialize(value_data);
329 if (value_opt) {
330 parsed[key] = std::move(*value_opt);
331 } else {
332 throw std::runtime_error("value_store::deserialize_binary() - failed to deserialize value for key: "
333 + key);
334 }
335 }
336
337 // Commit parsed entries atomically once the whole buffer is validated
338 std::unique_lock lock(store.mutex_);
339 store.values_ = std::move(parsed);
340}
