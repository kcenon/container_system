331 [[nodiscard]] kcenon::common::Result<T> get(std::string_view key) const noexcept {
332 std::shared_lock lock(mutex_);
333 auto result = storage_.get(key);
334 if (!result) {
335 return kcenon::common::Result<T>(
336 kcenon::common::error_info{
337 error_codes::key_not_found,
338 error_codes::make_message(error_codes::key_not_found, key),
339 "container_system"});
340 }
341 if (auto* ptr = std::get_if<T>(&result->data)) {
342 return kcenon::common::ok(*ptr);
343 }
344 return kcenon::common::Result<T>(
345 kcenon::common::error_info{
346 error_codes::type_mismatch,
347 error_codes::make_message(error_codes::type_mismatch, key),
348 "container_system"});
349 }
350
358 template<typename T>
359 [[nodiscard]] kcenon::common::VoidResult set_result(std::string_view key, T&& data_val) noexcept {
360 try {
361 if (key.empty()) {
362 return kcenon::common::VoidResult(
363 kcenon::common::error_info{
364 error_codes::empty_key,
365 error_codes::make_message(error_codes::empty_key),
366 "container_system"});
367 }
368
369 optimized_value val;
370 val.name = std::string(key);
371 val.data = std::forward<T>(data_val);
372 val.type = static_cast<value_types>(val.data.index());
373
374 std::unique_lock lock(mutex_);
375 storage_.set(key, std::move(val));
376 return kcenon::common::ok();
377 } catch (const std::bad_alloc&) {
378 return kcenon::common::VoidResult(
379 kcenon::common::error_info{
380 error_codes::memory_allocation_failed,
381 error_codes::make_message(error_codes::memory_allocation_failed, key),
382 "container_system"});
383 } catch (const std::exception& e) {
384 return kcenon::common::VoidResult(
385 kcenon::common::error_info{
386 error_codes::invalid_value,
387 std::string("Failed to set value: ") + e.what(),
388 "container_system"});
389 }
390 }

417 size_t reserve_hint = 0) {
418 std::unique_lock lock(mutex_);

433 [[nodiscard]] std::vector<std::optional<optimized_value>> get_batch(
434 std::span<const std::string_view> keys) const noexcept {
435 std::shared_lock lock(mutex_);
436 std::vector<std::optional<optimized_value>> results;
437 results.reserve(keys.size());
438 for (const auto& key : keys) {
439 results.push_back(storage_.get(key));
440 }
441 return results;
442 }
443
449 [[nodiscard]] std::vector<bool> contains_batch(
450 std::span<const std::string_view> keys) const noexcept {
451 std::shared_lock lock(mutex_);
452 std::vector<bool> results;
453 results.reserve(keys.size());
454 for (const auto& key : keys) {
455 results.push_back(storage_.contains(key));
456 }
457 return results;
458 }

465 size_t remove_batch(std::span<const std::string_view> keys) {
466 std::unique_lock lock(mutex_);
467 size_t count = 0;
468 for (const auto& key : keys) {
469 if (storage_.remove(key)) {
470 ++count;
471 }
472 }
473 return count;
474 }
