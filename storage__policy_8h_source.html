60template<typename P>
61concept StoragePolicy = requires(P p, const P cp, std::string_view key) {
62 // Type requirements
63 typename P::value_type;
64 typename P::iterator;
65 typename P::const_iterator;
66
67 // CRUD operations
68 { p.set(key, std::declval<const typename P::value_type&>()) } -> std::same_as<void>;
69 { cp.get(key) } -> std::same_as<std::optional<typename P::value_type>>;
70 { cp.contains(key) } -> std::convertible_to<bool>;
71 { p.remove(key) } -> std::convertible_to<bool>;
72 { p.clear() } -> std::same_as<void>;
73
74 // Iteration support
75 { p.begin() } -> std::same_as<typename P::iterator>;
76 { p.end() } -> std::same_as<typename P::iterator>;
77 { cp.begin() } -> std::same_as<typename P::const_iterator>;
78 { cp.end() } -> std::same_as<typename P::const_iterator>;
79 { cp.size() } -> std::convertible_to<std::size_t>;
80 { cp.empty() } -> std::convertible_to<bool>;
81};

307 void set(std::string_view key, const value_type& value) {
308 std::string key_str(key);
309 auto it = index_.find(key_str);
310 if (it != index_.end()) {
311 values_[it->second] = value;
312 } else {
313 index_[key_str] = values_.size();

315 }

323 void set(std::string_view key, value_type&& value) {
324 std::string key_str(key);
325 auto it = index_.find(key_str);
326 if (it != index_.end()) {
327 values_[it->second] = std::move(value);
328 } else {
329 index_[key_str] = values_.size();
330 values_.push_back(std::move(value));
331 }

339 [[nodiscard]] std::optional<value_type> get(std::string_view key) const {
340 auto it = index_.find(std::string(key));
341 if (it != index_.end()) {
342 return values_[it->second];
343 }
344 return std::nullopt;
345 }
346
352 [[nodiscard]] bool contains(std::string_view key) const {
353 return index_.find(std::string(key)) != index_.end();
354 }
355
363 bool remove(std::string_view key) {
364 std::string key_str(key);
365 auto it = index_.find(key_str);
366 if (it == index_.end()) {
367 return false;
368 }
369
370 std::size_t idx = it->second;
371 values_.erase(values_.begin() + static_cast<std::ptrdiff_t>(idx));
372 index_.erase(it);
373
374 // Rebuild index for elements after the removed one
375 for (auto& [k, v] : index_) {
376 if (v > idx) {
377 --v;
378 }
379 }
380 return true;
381 }
382
386 void clear() {
387 values_.clear();
388 index_.clear();
389 }
390
391 iterator begin() { return values_.begin(); }
392 iterator end() { return values_.end(); }
393 [[nodiscard]] const_iterator begin() const { return values_.begin(); }
394 [[nodiscard]] const_iterator end() const { return values_.end(); }
395 [[nodiscard]] std::size_t size() const { return values_.size(); }
396 [[nodiscard]] bool empty() const { return values_.empty(); }
397
402 void reserve(std::size_t capacity) {
403 values_.reserve(capacity);
404 index_.reserve(capacity);
405 }
406
412 void rebuild_index() {
413 index_.clear();
414 index_.reserve(values_.size());
415 for (std::size_t i = 0; i < values_.size(); ++i) {
416 index_[values_[i].name] = i;
417 }
418 }
419
420 container_type& data() { return values_; }
421 [[nodiscard]] const container_type& data() const { return values_; }
422
423private:
424 container_type values_;

476 template<typename T>
477 static constexpr bool is_allowed_type_v =
478 (std::is_same_v<std::decay_t<T>, AllowedTypes> || ...);

488 template<typename T>
489 void set_typed(std::string_view key, T&& typed_value) {
490 static_assert(is_allowed_type_v<T>,
491 "Type not in allowed types list for static_storage_policy");
492
493 value_type val;
494 val.name = std::string(key);
495 val.data = std::forward<T>(typed_value);
496 val.type = static_cast<value_types>(val.data.index());
497
498 auto it = find_by_key(key);
499 if (it != values_.end()) {
500 *it = std::move(val);
501 } else {
502 values_.push_back(std::move(val));
503 }
504 }
505
513 void set(std::string_view key, const value_type& value) {
514 if (!is_variant_type_allowed(value.data)) {
515 return; // Silently ignore disallowed types for interface compatibility
516 }
517
518 auto it = find_by_key(key);
519 if (it != values_.end()) {
520 *it = value;
521 } else {
522 values_.push_back(value);
523 }
524 }

530 if (!is_variant_type_allowed(value.data)) {
531 return;
532 }

534 auto it = find_by_key(key);
